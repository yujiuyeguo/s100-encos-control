cmake_minimum_required(VERSION 3.16)
project(s100_robot_hardware LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(encos_motors REQUIRED)
add_executable(s100_hardware_node src/hardware_node.cpp)
target_compile_features(s100_hardware_node PRIVATE cxx_std_17)
target_compile_options(s100_hardware_node PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(s100_hardware_node encos_motors::encos_motors)
ament_target_dependencies(s100_hardware_node rclcpp sensor_msgs)
install(TARGETS s100_hardware_node DESTINATION lib/${PROJECT_NAME})
ament_package()
