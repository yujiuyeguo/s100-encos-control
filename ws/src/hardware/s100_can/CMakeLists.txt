cmake_minimum_required(VERSION 3.16)
project(s100_can LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
option(S100_ENABLE_CANHAL "Build S100 vendor CANHAL backend" OFF)
add_library(${PROJECT_NAME} src/transport.cpp)
target_compile_features(${PROJECT_NAME} PUBLIC cxx_std_17)
target_compile_options(${PROJECT_NAME} PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(${PROJECT_NAME} PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
if(S100_ENABLE_CANHAL)
  find_path(CANHAL_INCLUDE_DIR hobot_can_hal.h PATHS /usr/hobot/include/canhal REQUIRED)
  find_library(CANHAL_LIBRARY hbcanhal PATHS /usr/hobot/lib REQUIRED)
  target_sources(${PROJECT_NAME} PRIVATE src/canhal_transport.cpp)
  target_compile_definitions(${PROJECT_NAME} PRIVATE S100_ENABLE_CANHAL)
  target_include_directories(${PROJECT_NAME} PRIVATE ${CANHAL_INCLUDE_DIR})
  target_link_libraries(${PROJECT_NAME} PRIVATE ${CANHAL_LIBRARY})
endif()
install(TARGETS ${PROJECT_NAME} EXPORT export_${PROJECT_NAME}
  ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(DIRECTORY include/ DESTINATION include)
if(BUILD_TESTING)
  enable_testing()
  add_executable(test_transport test/test_transport.cpp)
  target_link_libraries(test_transport PRIVATE ${PROJECT_NAME})
  add_test(NAME transport_contract COMMAND test_transport)
endif()
ament_export_targets(export_${PROJECT_NAME} HAS_LIBRARY_TARGET)
ament_package()
